  (defun my/latex-side-by-side ()
  "Display the current LaTeX source and its PDF side by side."
  (interactive)
  (let* ((tex-buffer (current-buffer))
         (pdf-file (concat (file-name-sans-extension
                            (buffer-file-name))
                           ".pdf")))
    (unless (file-exists-p pdf-file)
      (user-error "PDF not found. Compile your LaTeX file first"))

    (delete-other-windows)
    (split-window-right)

    ;; Keep LaTeX source on the left.
    (set-window-buffer (selected-window) tex-buffer)

    ;; Open PDF on the right.
    (other-window 1)
    (find-file pdf-file)

    ;; Return focus to LaTeX.
    (other-window -1)))

(use-package ispell
  :ensure nil
  :custom

(use-package flyspell
  :ensure nil
  :hook
  (LaTeX-mode . flyspell-mode)
  (org-mode . flyspell-mode)
  (text-mode . flyspell-mode)
  :bind
  ("C-c s f" . my/spell-french)
  ("C-c s e" . my/spell-english)
  ("C-c s t" . my/spell-toggle-language))

(defun my/spell-french ()
  "Use French spelling dictionary."
  (interactive)
  (ispell-change-dictionary "fr_FR")
  (when flyspell-mode (flyspell-buffer)))

(defun my/spell-english ()
  "Use English spelling dictionary."
  (interactive)
  (ispell-change-dictionary "en_US")
  (when flyspell-mode (flyspell-buffer)))

(defun my/spell-toggle-language ()
  "Toggle French and English dictionaries."
  (interactive)
  (if (equal (or ispell-local-dictionary ispell-dictionary)
             "fr_FR")
      (my/spell-english)
    (my/spell-french)))
